Programación Orientada a Objetos (POO)

Clases y Objetos

<?php
class Usuario {
    // Propiedades
    public $nombre;
    public $email;
    private $password;
    protected $rol;

    // Constructor
    public function __construct($nombre, $email, $password) {
        $this->nombre = $nombre;
        $this->email = $email;
        $this->setPassword($password);
    }

    // Métodos
    public function getNombre() {
        return $this->nombre;
    }

    public function setPassword($password) {
        $this->password = password_hash($password, PASSWORD_DEFAULT);
    }

    public function verificarPassword($password) {
        return password_verify($password, $this->password);
    }
}

// Crear objeto
$usuario = new Usuario("Juan Pérez", "juan@email.com", "123456");
echo $usuario->getNombre();
?>

Herencia

<?php
class Administrador extends Usuario {
    public function __construct($nombre, $email, $password) {
        parent::__construct($nombre, $email, $password);
        $this->rol = "admin";
    }

    public function banearUsuario($usuarioId) {
        // Lógica para banear usuario
        return "Usuario $usuarioId baneado";
    }
}

$admin = new Administrador("Admin", "admin@email.com", "admin123");
echo $admin->getNombre(); // Heredado de Usuario
echo $admin->banearUsuario(123);
?>

MySQL con MySQLi

Conexión a la Base de Datos

<?php
// Conexión
$conexion = new mysqli("localhost", "usuario", "password", "base_datos");

// Verificar conexión
if ($conexion->connect_error) {
    die("Error de conexión: " . $conexion->connect_error);
}

// Configurar charset
$conexion->set_charset("utf8");
?>

Consultas Básicas

<?php
// INSERT
$sql = "INSERT INTO usuarios (nombre, email) VALUES (?, ?)";
$stmt = $conexion->prepare($sql);
$stmt->bind_param("ss", $nombre, $email);
$stmt->execute();

// SELECT
$sql = "SELECT id, nombre, email FROM usuarios WHERE activo = 1";
$resultado = $conexion->query($sql);

if ($resultado->num_rows > 0) {
    while($fila = $resultado->fetch_assoc()) {
        echo "ID: " . $fila["id"] . " - Nombre: " . $fila["nombre"] . "<br>";
    }
}

// UPDATE
$sql = "UPDATE usuarios SET nombre = ? WHERE id = ?";
$stmt = $conexion->prepare($sql);
$stmt->bind_param("si", $nuevoNombre, $id);
$stmt->execute();

// DELETE
$sql = "DELETE FROM usuarios WHERE id = ?";
$stmt = $conexion->prepare($sql);
$stmt->bind_param("i", $id);
$stmt->execute();

$conexion->close();
?>

MySQL con PDO

Conexión PDO

<?php
try {
    $pdo = new PDO("mysql:host=localhost;dbname=base_datos;charset=utf8", "usuario", "password");
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
} catch(PDOException $e) {
    die("Error de conexión: " . $e->getMessage());
}
?>

Operaciones CRUD con PDO

<?php
// INSERT
$sql = "INSERT INTO usuarios (nombre, email, fecha_registro) VALUES (:nombre, :email, NOW())";
$stmt = $pdo->prepare($sql);
$stmt->execute([
    'nombre' => $nombre,
    'email' => $email
]);

// SELECT
$sql = "SELECT * FROM usuarios WHERE activo = :activo ORDER BY fecha_registro DESC";
$stmt = $pdo->prepare($sql);
$stmt->execute(['activo' => 1]);
$usuarios = $stmt->fetchAll(PDO::FETCH_ASSOC);

foreach ($usuarios as $usuario) {
    echo $usuario['nombre'] . "<br>";
}

// UPDATE
$sql = "UPDATE usuarios SET nombre = :nombre WHERE id = :id";
$stmt = $pdo->prepare($sql);
$stmt->execute([
    'nombre' => $nuevoNombre,
    'id' => $id
]);

// DELETE
$sql = "DELETE FROM usuarios WHERE id = :id";
$stmt = $pdo->prepare($sql);
$stmt->execute(['id' => $id]);
?>
