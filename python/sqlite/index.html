📊 SQLite - Base de Datos Ligera

Introducción a SQLite

SQLite es una base de datos relacional ligera que no requiere servidor. Es perfecta para desarrollo, prototipos y aplicaciones pequeñas.

¿Por qué SQLite?

  • Sin servidor: No requiere instalación de servidor de base de datos
  • Archivo único: Toda la base de datos en un solo archivo
  • Zero-config: No necesita configuración
  • Portable: Fácil de mover entre sistemas
  • Rápido: Excelente performance para operaciones locales
  • SQL estándar: Compatible con la mayoría de comandos SQL

Casos de uso ideales

  • Aplicaciones de escritorio
  • Prototipos y desarrollo
  • Aplicaciones móviles
  • Datos embebidos
  • Testing y desarrollo local
  • Configuraciones y caches locales

Joins

-- Crear tablas de ejemplo
CREATE TABLE departamentos (
    id INTEGER PRIMARY KEY,
    nombre TEXT NOT NULL,
    ubicacion TEXT
);

CREATE TABLE empleados (
    id INTEGER PRIMARY KEY,
    nombre TEXT NOT NULL,
    departamento_id INTEGER,
    salario REAL,
    FOREIGN KEY (departamento_id) REFERENCES departamentos(id)
);

-- INNER JOIN
SELECT e.nombre, d.nombre as departamento, e.salario
FROM empleados e
INNER JOIN departamentos d ON e.departamento_id = d.id;

-- LEFT JOIN
SELECT e.nombre, d.nombre as departamento
FROM empleados e
LEFT JOIN departamentos d ON e.departamento_id = d.id;

-- RIGHT JOIN (usando LEFT JOIN invertido)
SELECT d.nombre as departamento, e.nombre
FROM departamentos d
LEFT JOIN empleados e ON d.id = e.departamento_id;

-- FULL OUTER JOIN (simulado con UNION)
SELECT e.nombre, d.nombre as departamento
FROM empleados e
LEFT JOIN departamentos d ON e.departamento_id = d.id
UNION ALL
SELECT NULL, d.nombre
FROM departamentos d
LEFT JOIN empleados e ON d.id = e.departamento_id
WHERE e.id IS NULL;

-- JOIN con múltiples tablas
SELECT e.nombre, d.nombre as departamento, p.nombre as proyecto
FROM empleados e
JOIN departamentos d ON e.departamento_id = d.id
JOIN proyectos_empleados pe ON e.id = pe.empleado_id
JOIN proyectos p ON pe.proyecto_id = p.id;

Subconsultas

-- Subconsulta en WHERE
SELECT nombre, salario FROM empleados 
WHERE salario > (SELECT AVG(salario) FROM empleados);

-- Subconsulta en SELECT
SELECT nombre, 
       salario,
       (SELECT nombre FROM departamentos WHERE id = empleados.departamento_id) as departamento
FROM empleados;

-- Subconsulta en FROM
SELECT departamento, COUNT(*) as empleados, AVG(salario) as salario_promedio
FROM (
    SELECT d.nombre as departamento, e.salario
    FROM empleados e
    JOIN departamentos d ON e.departamento_id = d.id
) 
GROUP BY departamento;

-- EXISTS
SELECT nombre FROM empleados e
WHERE EXISTS (
    SELECT 1 FROM proyectos_empleados pe 
    WHERE pe.empleado_id = e.id
);

-- IN con subconsulta
SELECT nombre FROM empleados 
WHERE departamento_id IN (
    SELECT id FROM departamentos WHERE ubicacion = 'Madrid'
);

Transacciones

-- Iniciar transacción
BEGIN TRANSACTION;

-- Operaciones
UPDATE cuentas SET saldo = saldo - 100 WHERE id = 1;
UPDATE cuentas SET saldo = saldo + 100 WHERE id = 2;

-- Verificar consistencia
SELECT SUM(saldo) FROM cuentas;

-- Confirmar transacción
COMMIT;

-- O revertir si hay error
ROLLBACK;

-- Savepoints
BEGIN TRANSACTION;
UPDATE cuentas SET saldo = saldo - 50 WHERE id = 1;
SAVEPOINT punto_guardado;
UPDATE cuentas SET saldo = saldo - 30 WHERE id = 1;  -- Esto podría fallar
ROLLBACK TO SAVEPOINT punto_guardado;  -- Volver al savepoint
COMMIT;

Módulo sqlite3

import sqlite3
from datetime import datetime

def crear_conexion():
    """Crear conexión a la base de datos"""
    try:
        conn = sqlite3.connect('mi_base.db')
        print("Conexión exitosa a SQLite")
        return conn
    except sqlite3.Error as e:
        print(f"Error al conectar: {e}")
        return None

def crear_tabla(conn):
    """Crear tabla"""
    try:
        cursor = conn.cursor()
        
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS usuarios (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                nombre TEXT NOT NULL,
                email TEXT UNIQUE NOT NULL,
                edad INTEGER,
                activo BOOLEAN DEFAULT 1,
                fecha_registro TEXT DEFAULT CURRENT_TIMESTAMP
            )
        ''')
        
        conn.commit()
        print("Tabla creada exitosamente")
        
    except sqlite3.Error as e:
        print(f"Error al crear tabla: {e}")

def insertar_usuario(conn, usuario):
    """Insertar un usuario"""
    sql = '''INSERT INTO usuarios (nombre, email, edad) 
             VALUES (?, ?, ?)'''
    
    try:
        cursor = conn.cursor()
        cursor.execute(sql, usuario)
        conn.commit()
        
        print(f"Usuario insertado con ID: {cursor.lastrowid}")
        return cursor.lastrowid
        
    except sqlite3.Error as e:
        print(f"Error al insertar: {e}")
        return None

def consultar_usuarios(conn):
    """Consultar todos los usuarios"""
    try:
        cursor = conn.cursor()
        cursor.execute("SELECT * FROM usuarios")
        
        filas = cursor.fetchall()
        
        print("=== USUARIOS ===")
        for fila in filas:
            print(f"ID: {fila[0]}, Nombre: {fila[1]}, Email: {fila[2]}, Edad: {fila[3]}")
            
        return filas
        
    except sqlite3.Error as e:
        print(f"Error al consultar: {e}")
        return []

def actualizar_usuario(conn, usuario_id, nuevos_datos):
    """Actualizar usuario"""
    sql = '''UPDATE usuarios 
             SET nombre = ?, email = ?, edad = ? 
             WHERE id = ?'''
    
    try:
        cursor = conn.cursor()
        cursor.execute(sql, (*nuevos_datos, usuario_id))
        conn.commit()
        
        print(f"Usuario {usuario_id} actualizado")
        return cursor.rowcount
        
    except sqlite3.Error as e:
        print(f"Error al actualizar: {e}")
        return 0

def eliminar_usuario(conn, usuario_id):
    """Eliminar usuario"""
    sql = 'DELETE FROM usuarios WHERE id = ?'
    
    try:
        cursor = conn.cursor()
        cursor.execute(sql, (usuario_id,))
        conn.commit()
        
        print(f"Usuario {usuario_id} eliminado")
        return cursor.rowcount
        
    except sqlite3.Error as e:
        print(f"Error al eliminar: {e}")
        return 0

def main():
    # Crear conexión
    conn = crear_conexion()
    
    if conn is not None:
        # Crear tabla
        crear_tabla(conn)
        
        # Insertar usuarios
        usuarios = [
            ('Juan Pérez', 'juan@email.com', 25),
            ('María García', 'maria@email.com', 30),
            ('Carlos López', 'carlos@email.com', 28)
        ]
        
        for usuario in usuarios:
            insertar_usuario(conn, usuario)
        
        # Consultar usuarios
        consultar_usuarios(conn)
        
        # Actualizar usuario
        actualizar_usuario(conn, 1, ('Juan Carlos Pérez', 'juancarlos@email.com', 26))
        
        # Consultar nuevamente
        consultar_usuarios(conn)
        
        # Cerrar conexión
        conn.close()

Context Manager y Manejo de Errores

import sqlite3
import logging

# Configurar logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class DatabaseManager:
    def __init__(self, db_path):
        self.db_path = db_path
    
    def __enter__(self):
        self.conn = sqlite3.connect(self.db_path)
        self.conn.execute("PRAGMA foreign_keys = ON")  # Habilitar claves foráneas
        return self.conn
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.conn:
            if exc_type is None:
                self.conn.commit()
                logger.info("Transacción confirmada")
            else:
                self.conn.rollback()
                logger.error(f"Transacción revertida: {exc_val}")
            self.conn.close()

def crear_tabla_productos():
    """Crear tabla de productos con manejo de errores"""
    with DatabaseManager('tienda.db') as conn:
        try:
            conn.execute('''
                CREATE TABLE IF NOT EXISTS productos (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    nombre TEXT NOT NULL,
                    precio REAL NOT NULL CHECK(precio > 0),
                    stock INTEGER DEFAULT 0 CHECK(stock >= 0),
                    categoria TEXT,
                    fecha_creacion TEXT DEFAULT CURRENT_TIMESTAMP
                )
            ''')
            logger.info("Tabla productos creada")
        except sqlite3.Error as e:
            logger.error(f"Error creando tabla: {e}")
            raise

def insertar_productos(productos):
    """Insertar múltiples productos con transacción"""
    with DatabaseManager('tienda.db') as conn:
        try:
            conn.executemany('''
                INSERT INTO productos (nombre, precio, stock, categoria)
                VALUES (?, ?, ?, ?)
            ''', productos)
            logger.info(f"{len(productos)} productos insertados")
        except sqlite3.IntegrityError as e:
            logger.error(f"Error de integridad: {e}")
            raise
        except sqlite3.Error as e:
            logger.error(f"Error insertando productos: {e}")
            raise

def consultar_productos_con_paginacion(pagina=1, por_pagina=10):
    """Consultar productos con paginación"""
    offset = (pagina - 1) * por_pagina
    
    with DatabaseManager('tienda.db') as conn:
        try:
            cursor = conn.cursor()
            
            # Obtener productos
            cursor.execute('''
                SELECT id, nombre, precio, stock, categoria, fecha_creacion
                FROM productos
                ORDER BY fecha_creacion DESC
                LIMIT ? OFFSET ?
            ''', (por_pagina, offset))
            
            productos = cursor.fetchall()
            
            # Obtener total de productos
            cursor.execute('SELECT COUNT(*) FROM productos')
            total = cursor.fetchone()[0]
            
            return {
                'productos': productos,
                'total': total,
                'pagina': pagina,
                'por_pagina': por_pagina,
                'paginas_totales': (total + por_pagina - 1) // por_pagina
            }
            
        except sqlite3.Error as e:
            logger.error(f"Error consultando productos: {e}")
            return None

def buscar_productos(query, categoria=None, precio_max=None):
    """Búsqueda avanzada de productos"""
    with DatabaseManager('tienda.db') as conn:
        try:
            cursor = conn.cursor()
            
            sql = '''
                SELECT id, nombre, precio, stock, categoria
                FROM productos
                WHERE nombre LIKE ?
            '''
            params = [f'%{query}%']
            
            if categoria:
                sql += ' AND categoria = ?'
                params.append(categoria)
            
            if precio_max:
                sql += ' AND precio <= ?'
                params.append(precio_max)
            
            sql += ' ORDER BY precio ASC'
            
            cursor.execute(sql, params)
            return cursor.fetchall()
            
        except sqlite3.Error as e:
            logger.error(f"Error en búsqueda: {e}")
            return []

def actualizar_stock(producto_id, nuevo_stock):
    """Actualizar stock con validación"""
    if nuevo_stock < 0:
        raise ValueError("El stock no puede ser negativo")
    
    with DatabaseManager('tienda.db') as conn:
        try:
            cursor = conn.cursor()
            
            cursor.execute('''
                UPDATE productos 
                SET stock = ? 
                WHERE id = ?
            ''', (nuevo_stock, producto_id))
            
            if cursor.rowcount == 0:
                raise ValueError(f"Producto con ID {producto_id} no encontrado")
            
            logger.info(f"Stock del producto {producto_id} actualizado a {nuevo_stock}")
            
        except sqlite3.Error as e:
            logger.error(f"Error actualizando stock: {e}")
            raise

ORM con SQLAlchemy

from sqlalchemy import create_engine, Column, Integer, String, Float, Boolean, DateTime, ForeignKey
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, relationship
from datetime import datetime

# Crear engine para SQLite
engine = create_engine('sqlite:///tienda_orm.db', echo=True)

# Base para modelos
Base = declarative_base()

# Modelo de Categoría
class Categoria(Base):
    __tablename__ = 'categorias'
    
    id = Column(Integer, primary_key=True)
    nombre = Column(String(100), unique=True, nullable=False)
    descripcion = Column(String(255))
    
    # Relación con productos
    productos = relationship('Producto', back_populates='categoria')
    
    def __repr__(self):
        return f'<Categoria {self.nombre}>'

# Modelo de Producto
class Producto(Base):
    __tablename__ = 'productos'
    
    id = Column(Integer, primary_key=True)
    nombre = Column(String(200), nullable=False)
    precio = Column(Float, nullable=False)
    stock = Column(Integer, default=0)
    disponible = Column(Boolean, default=True)
    fecha_creacion = Column(DateTime, default=datetime.utcnow)
    
    # Clave foránea
    categoria_id = Column(Integer, ForeignKey('categorias.id'))
    
    # Relación
    categoria = relationship('Categoria', back_populates='productos')
    
    def __repr__(self):
        return f'<Producto {self.nombre} - ${self.precio}>'

# Crear tablas
Base.metadata.create_all(engine)

# Crear sesión
Session = sessionmaker(bind=engine)
session = Session()

def crear_datos_ejemplo():
    """Crear datos de ejemplo"""
    # Crear categorías
    electronica = Categoria(nombre='Electrónica', descripcion='Productos electrónicos')
    ropa = Categoria(nombre='Ropa', descripcion='Ropa y accesorios')
    
    session.add_all([electronica, ropa])
    session.commit()
    
    # Crear productos
    productos = [

        Producto(nombre='Camiseta', precio=19.99, stock=50, categoria=ropa),
        Producto(nombre='Libro Python', precio=39.99, stock=30, categoria=ropa)
    ]
    
    session.add_all(productos)
    session.commit()

def consultar_productos():
    """Consultar productos con sus categorías"""
    productos = session.query(Producto).join(Categoria).all()
    
    for producto in productos:
        print(f"{producto.nombre} - ${producto.precio} ({producto.categoria.nombre})")

def buscar_por_categoria(nombre_categoria):
    """Buscar productos por categoría"""
    productos = session.query(Producto).join(Categoria)\
        .filter(Categoria.nombre == nombre_categoria).all()
    
    return productos

def actualizar_precio(producto_id, nuevo_precio):
    """Actualizar precio de producto"""
    producto = session.query(Producto).get(producto_id)
    if producto:
        producto.precio = nuevo_precio
        session.commit()
        print(f"Precio de {producto.nombre} actualizado a ${nuevo_precio}")

def eliminar_producto(producto_id):
    """Eliminar producto"""
    producto = session.query(Producto).get(producto_id)
    if producto:
        session.delete(producto)
        session.commit()
        print(f"Producto {producto.nombre} eliminado")
