Introducción a Node.js

Node.js es un entorno de ejecución de JavaScript del lado del servidor que permite crear aplicaciones web escalables. Está construido sobre el motor V8 de Chrome y utiliza un modelo de I/O no bloqueante.

¿Por qué Node.js?

  • JavaScript Full-Stack: Mismo lenguaje en frontend y backend
  • Alto rendimiento: I/O no bloqueante y event-driven
  • Gran ecosistema: NPM con millones de paquetes
  • Escalabilidad: Manejo eficiente de conexiones concurrentes
  • Comunidad activa: Actualizaciones constantes y soporte

Servidor HTTP Básico

const http = require('http');

// Crear servidor
const server = http.createServer((req, res) => {
    // Configurar headers
    res.writeHead(200, {
        'Content-Type': 'text/html; charset=utf-8'
    });

    // Routing básico
    if (req.url === '/') {
        res.end('
Inicio
');
    } else if (req.url === '/about') {
        res.end('
Acerca de
');
    } else {
        res.writeHead(404);
        res.end('
404 - Página no encontrada
');
    }
});

// Puerto del servidor
const PORT = process.env.PORT || 3000;

// Iniciar servidor
server.listen(PORT, () => {
    console.log(`Servidor corriendo en http://localhost:${PORT}`);
});

const express = require('express');
const app = express();

// Middleware para parsear JSON
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

app.use((req, res, next) => {
    console.log(`${req.method} ${req.path} - ${new Date().toISOString()}`);
    next();

// Rutas
app.get('/', (req, res) => {
    res.json({ mensaje: 'API funcionando' });
});

app.get('/api/usuarios', (req, res) => {
    // Simular datos
    const usuarios = [
        { id: 1, nombre: 'Juan', email: 'juan@email.com' },
        { id: 2, nombre: 'María', email: 'maria@email.com' }
    ];
    res.json(usuarios);
});

app.post('/api/usuarios', (req, res) => {
    const { nombre, email } = req.body;

    if (!nombre || !email) {
        return res.status(400).json({ error: 'Nombre y email requeridos' });
    }

    const nuevoUsuario = {
        id: Date.now(),
        nombre,
        email
    };

    res.status(201).json(nuevoUsuario);

app.put('/api/usuarios/:id', (req, res) => {
    const { id } = req.params;
    const { nombre, email } = req.body;

    // Lógica para actualizar usuario

app.delete('/api/usuarios/:id', (req, res) => {
    const { id } = req.params;
    // Lógica para eliminar usuario
    res.json({ mensaje: `Usuario ${id} eliminado` });

Middleware Avanzado

// middleware/auth.js
const jwt = require('jsonwebtoken');

const authenticateToken = (req, res, next) => {
    const authHeader = req.headers['authorization'];
    const token = authHeader && authHeader.split(' ')[1];

    if (!token) {
        return res.status(401).json({ error: 'Token requerido' });
    }

    jwt.verify(token, process.env.JWT_SECRET, (err, user) => {
        if (err) {
            return res.status(403).json({ error: 'Token inválido' });
        }
        req.user = user;
        next();
    });
};

module.exports = { authenticateToken };

// Uso en rutas
const { authenticateToken } = require('./middleware/auth');

app.get('/api/profile', authenticateToken, (req, res) => {
    res.json({ user: req.user });
});

Programación Asíncrona

Callbacks

// Callback hell (evitar)
fs.readFile('archivo1.txt', 'utf8', (err, data1) => {
    if (err) return console.error(err);
    
    fs.readFile('archivo2.txt', 'utf8', (err, data2) => {
        if (err) return console.error(err);
        
        fs.writeFile('combinado.txt', data1 + data2, (err) => {
            if (err) return console.error(err);
            console.log('Archivos combinados');
        });
    });
});

Promises

const fs = require('fs').promises;

const leerArchivo = (ruta) => {
    return fs.readFile(ruta, 'utf8');
};

const escribirArchivo = (ruta, contenido) => {
    return fs.writeFile(ruta, contenido);
};

// Usando Promises
leerArchivo('archivo1.txt')
    .then(data1 => {
        return leerArchivo('archivo2.txt')
            .then(data2 => {
                return escribirArchivo('combinado.txt', data1 + data2);
            });
    })
    .then(() => {
        console.log('Archivos combinados exitosamente');
    })
    .catch(err => {
        console.error('Error:', err);
    });

// Promise.all para operaciones paralelas
Promise.all([
    leerArchivo('archivo1.txt'),
    leerArchivo('archivo2.txt'),
    leerArchivo('archivo3.txt')
])
.then(resultados => {
    const contenidoTotal = resultados.join('\n');
    return escribirArchivo('todos.txt', contenidoTotal);
})
.then(() => console.log('Todos los archivos combinados'))
.catch(err => console.error(err));

Async/Await

const fs = require('fs').promises;

async function procesarArchivos() {
    try {
        const data1 = await fs.readFile('archivo1.txt', 'utf8');
        const data2 = await fs.readFile('archivo2.txt', 'utf8');
        
        const contenidoCombinado = data1 + '\n' + data2;
        await fs.writeFile('combinado.txt', contenidoCombinado);
        
        console.log('Archivos procesados exitosamente');
    } catch (error) {
        console.error('Error procesando archivos:', error);
    }
}

// Ejecutar función async
procesarArchivos();

// Con Promise.all para paralelismo
async function procesarArchivosParalelo() {
    try {
        const [data1, data2, data3] = await Promise.all([
            fs.readFile('archivo1.txt', 'utf8'),
            fs.readFile('archivo2.txt', 'utf8'),
            fs.readFile('archivo3.txt', 'utf8')
        ]);
        
        const contenidoTotal = [data1, data2, data3].join('\n---\n');
        await fs.writeFile('todos.txt', contenidoTotal);
        
        console.log('Todos los archivos procesados');
    } catch (error) {
        console.error('Error:', error);
    }
}

API de Tareas

// server.js
const express = require('express');
const jwt = require('jsonwebtoken');
const bcrypt = require('bcrypt');
const cors = require('cors');
require('dotenv').config();

const app = express();

// Middleware
app.use(cors());
app.use(express.json());

// Simulación de base de datos
let tareas = [];
let usuarios = [];

// Middleware de autenticación
const authenticateToken = (req, res, next) => {
    const token = req.header('Authorization')?.replace('Bearer ', '');
    
    if (!token) {
        return res.status(401).json({ error: 'Acceso denegado' });
    }
    
    try {
        const verified = jwt.verify(token, process.env.JWT_SECRET);
        req.user = verified;
        next();
    } catch (error) {
        res.status(400).json({ error: 'Token inválido' });
    }
};

// Rutas de autenticación
app.post('/register', async (req, res) => {
    try {
        const { username, password } = req.body;
        
        // Verificar si usuario existe
        const usuarioExistente = usuarios.find(u => u.username === username);
        if (usuarioExistente) {
            return res.status(400).json({ error: 'Usuario ya existe' });
        }
        
        // Hash de contraseña
        const salt = await bcrypt.genSalt(10);
        const hashedPassword = await bcrypt.hash(password, salt);
        
        // Crear usuario
        const usuario = {
            id: usuarios.length + 1,
            username,
            password: hashedPassword
        };
        
        usuarios.push(usuario);
        res.status(201).json({ message: 'Usuario registrado' });
    } catch (error) {
        res.status(500).json({ error: 'Error en el servidor' });
    }
});

app.post('/login', async (req, res) => {
    try {
        const { username, password } = req.body;
        
        // Buscar usuario
        const usuario = usuarios.find(u => u.username === username);
        if (!usuario) {
            return res.status(400).json({ error: 'Usuario no encontrado' });
        }
        
        // Verificar contraseña
        const validPassword = await bcrypt.compare(password, usuario.password);
        if (!validPassword) {
            return res.status(400).json({ error: 'Contraseña incorrecta' });
        }
        
        // Crear token
        const token = jwt.sign(
            { id: usuario.id, username: usuario.username },
            process.env.JWT_SECRET,
            { expiresIn: '1h' }
        );
        
        res.json({ token });
    } catch (error) {
        res.status(500).json({ error: 'Error en el servidor' });
    }
});

// Rutas de tareas (protegidas)
app.get('/tareas', authenticateToken, (req, res) => {
    const tareasUsuario = tareas.filter(t => t.usuarioId === req.user.id);
    res.json(tareasUsuario);
});

app.post('/tareas', authenticateToken, (req, res) => {
    const { titulo, descripcion } = req.body;
    
    const nuevaTarea = {
        id: tareas.length + 1,
        titulo,
        descripcion,
        completada: false,
        usuarioId: req.user.id,
        fechaCreacion: new Date()
    };
    
    tareas.push(nuevaTarea);
    res.status(201).json(nuevaTarea);
});

app.put('/tareas/:id', authenticateToken, (req, res) => {
    const { id } = req.params;
    const { titulo, descripcion, completada } = req.body;
    
    const tareaIndex = tareas.findIndex(t => t.id == id && t.usuarioId === req.user.id);
    
    if (tareaIndex === -1) {
        return res.status(404).json({ error: 'Tarea no encontrada' });
    }
    
    tareas[tareaIndex] = {
        ...tareas[tareaIndex],
        titulo: titulo || tareas[tareaIndex].titulo,
        descripcion: descripcion || tareas[tareaIndex].descripcion,
        completada: completada !== undefined ? completada : tareas[tareaIndex].completada
    };
    
    res.json(tareas[tareaIndex]);
});

app.delete('/tareas/:id', authenticateToken, (req, res) => {
    const { id } = req.params;
    
    const tareaIndex = tareas.findIndex(t => t.id == id && t.usuarioId === req.user.id);
    
    if (tareaIndex === -1) {
        return res.status(404).json({ error: 'Tarea no encontrada' });
    }
    
    tareas.splice(tareaIndex, 1);
    res.json({ message: 'Tarea eliminada' });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
    console.log(`Servidor corriendo en puerto ${PORT}`);
});
