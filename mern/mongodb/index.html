🍃 MongoDB - Base de Datos NoSQL

Introducción a MongoDB

MongoDB es una base de datos NoSQL orientada a documentos que almacena datos en formato BSON (JSON binario). Es altamente escalable y flexible.

¿Por qué MongoDB?

  • Esquema flexible: No requiere estructura fija de tablas
  • Escalable horizontalmente: Sharding automático
  • Alta performance: Índices eficientes y consultas rápidas
  • JSON-like: Documentos similares a objetos JavaScript
  • Replicación: Alta disponibilidad con replica sets
  • Rich queries: Consultas avanzadas y agregaciones

Conceptos Básicos

  • Documento: Unidad básica de datos (similar a fila en SQL)
  • Colección: Grupo de documentos (similar a tabla en SQL)
  • Base de datos: Contenedor de colecciones
  • BSON: Formato binario para almacenar documentos
  • ObjectId: Identificador único de 12 bytes

        this.client = new MongoClient(process.env.MONGODB_URI);
        this.dbName = process.env.DB_NAME;
        this.db = null;

        try {
            await this.client.connect();
            this.db = this.client.db(this.dbName);

            console.error('Error de conexión:', error);

        }

        return this.db;
    }

    async close() {
        await this.client.close();
        console.log('Conexión cerrada');
    }
}

    try {
        await client.connect();

        await client.close();

Framework de Agregación

Pipeline de Agregación

// $match - Filtrar documentos
db.ventas.aggregate([
    { $match: { fecha: { $gte: ISODate("2024-01-01") } } }
])

// $group - Agrupar documentos
db.ventas.aggregate([
    {
        $group: {
            _id: "$producto",
            totalVentas: { $sum: "$cantidad" },
            ingresosTotales: { $sum: { $multiply: ["$cantidad", "$precio"] } },
            promedioPrecio: { $avg: "$precio" }
        }
    }
])

// $project - Reformatear documentos
db.usuarios.aggregate([
    {
        $project: {
            nombreCompleto: { $concat: ["$nombre", " ", "$apellido"] },
            edad: 1,
            email: 1,
            _id: 0
        }
    }
])

// $sort - Ordenar
db.productos.aggregate([
    { $sort: { precio: -1, nombre: 1 } }
])

// $limit y $skip - Paginación
db.productos.aggregate([
    { $sort: { precio: -1 } },
    { $skip: 10 },
    { $limit: 5 }
])

// Pipeline completo
db.pedidos.aggregate([
    // 1. Filtrar pedidos del último mes
    {
        $match: {
            fecha: { $gte: new Date(Date.now() - 30*24*60*60*1000) },
            estado: "completado"
        }
    },
    
    // 2. Unir con colección de usuarios
    {
        $lookup: {
            from: "usuarios",
            localField: "usuarioId",
            foreignField: "_id",
            as: "usuario"
        }
    },
    
    // 3. Desenrollar array de usuario
    { $unwind: "$usuario" },
    
    // 4. Agrupar por ciudad
    {
        $group: {
            _id: "$usuario.direccion.ciudad",
            totalPedidos: { $sum: 1 },
            ingresosTotales: { $sum: "$total" },
            clientesUnicos: { $addToSet: "$usuarioId" }
        }
    },
    
    // 5. Calcular número de clientes únicos
    {
        $project: {
            ciudad: "$_id",
            totalPedidos: 1,
            ingresosTotales: 1,
            numClientes: { $size: "$clientesUnicos" },
            promedioPorCliente: { $divide: ["$ingresosTotales", { $size: "$clientesUnicos" }] }
        }
    },
    
    // 6. Ordenar por ingresos
    { $sort: { ingresosTotales: -1 } }
])

Esquemas y Modelos

const mongoose = require('mongoose');

// Esquema de Usuario
const usuarioSchema = new mongoose.Schema({
    nombre: {
        type: String,
        required: [true, 'El nombre es obligatorio'],
        trim: true,
        maxlength: [50, 'El nombre no puede exceder 50 caracteres']
    },
    email: {
        type: String,
        required: [true, 'El email es obligatorio'],
        unique: true,
        lowercase: true,
        validate: {
            validator: function(email) {
                return /^\w+([\.-]?\w+)*@\w+([\.-]?\w+)*(\.\w{2,3})+$/.test(email);
            },
            message: 'Email inválido'
        }
    },
    password: {
        type: String,
        required: [true, 'La contraseña es obligatoria'],
        minlength: [6, 'La contraseña debe tener al menos 6 caracteres'],
        select: false // No incluir en consultas por defecto
    },
    edad: {
        type: Number,
        min: [0, 'La edad no puede ser negativa'],
        max: [120, 'Edad no válida']
    },
    activo: {
        type: Boolean,
        default: true
    },
    rol: {
        type: String,
        enum: ['usuario', 'moderador', 'admin'],
        default: 'usuario'
    },
    fechaRegistro: {
        type: Date,
        default: Date.now
    },
    ultimoLogin: Date,
    avatar: String,
    hobbies: [String],
    direccion: {
        calle: String,
        ciudad: String,
        pais: {
            type: String,
            default: 'España'
        }
    }
}, {
    timestamps: true, // Agrega createdAt y updatedAt
    toJSON: { virtuals: true },
    toObject: { virtuals: true }
});

// Índices
usuarioSchema.index({ email: 1 });
usuarioSchema.index({ nombre: 1, apellido: 1 });

// Virtual para nombre completo
usuarioSchema.virtual('nombreCompleto').get(function() {
    return `${this.nombre}`;
});

// Método de instancia
usuarioSchema.methods.verificarPassword = function(password) {
    return this.password === password; // En producción usar bcrypt
};

// Método estático
usuarioSchema.statics.buscarPorEmail = function(email) {
    return this.findOne({ email: email.toLowerCase() });
};

// Middleware pre-save
usuarioSchema.pre('save', function(next) {
    if (this.isModified('password')) {
        // En producción hashear contraseña
        console.log('Password modificada, hasheando...');
    }
    next();
});

// Crear modelo
const Usuario = mongoose.model('Usuario', usuarioSchema);
